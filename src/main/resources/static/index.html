const form = document.getElementById('formProduto'); const listaProdutos = document.getElementById('listaProdutos'); const mensagemEl = document.getElementById('mensagem'); const tituloFormulario = document.getElementById('tituloFormulario'); const btnCancelar = document.getElementById('btnCancelar'); let editandoId = null; function mostrarMensagem(texto, sucesso = true) { mensagemEl.textContent = texto; mensagemEl.style.color = sucesso ? '#1e8e5a' : '#b5121b'; } function limparFormulario() { form.reset(); document.getElementById('idProduto').value = ''; document.getElementById('categoria').value = 'Smartphones'; editandoId = null; tituloFormulario.textContent = 'Cadastrar produto'; } function formatarMoeda(valor) { return Number(valor).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }); } function montarItem(produto) { const item = document.createElement('article'); item.className = 'item-admin'; item.innerHTML = `
${produto.nome}
${produto.marca} • ${produto.categoria}
${formatarMoeda(produto.preco)}

${produto.descricao}

Estoque: ${produto.estoque} ID: ${produto.id}
`; return item; } async function carregarProdutos() { try { const resposta = await fetch('/api/produtos'); if (!resposta.ok) { throw new Error('Erro ao carregar os produtos.'); } const produtos = await resposta.json(); listaProdutos.innerHTML = ''; produtos.forEach((produto) => { listaProdutos.appendChild(montarItem(produto)); }); listaProdutos.querySelectorAll('[data-editar]').forEach((botao) => { botao.addEventListener('click', async () => { const id = botao.getAttribute('data-editar'); const resposta = await fetch(`/api/produtos/${id}`); const produto = await resposta.json(); document.getElementById('idProduto').value = produto.id; document.getElementById('nome').value = produto.nome; document.getElementById('descricao').value = produto.descricao; document.getElementById('preco').value = produto.preco; document.getElementById('estoque').value = produto.estoque; document.getElementById('marca').value = produto.marca; document.getElementById('categoria').value = produto.categoria; document.getElementById('imagem').value = produto.imagem; editandoId = produto.id; tituloFormulario.textContent = 'Editar produto'; window.scrollTo({ top: 0, behavior: 'smooth' }); }); }); listaProdutos.querySelectorAll('[data-excluir]').forEach((botao) => { botao.addEventListener('click', async () => { const id = botao.getAttribute('data-excluir'); const confirmar = confirm('Deseja realmente excluir este produto?'); if (!confirmar) return; const resposta = await fetch(`/api/produtos/${id}`, { method: 'DELETE' }); if (resposta.ok) { mostrarMensagem('Produto excluído com sucesso!'); carregarProdutos(); } else { mostrarMensagem('Erro ao excluir produto.', false); } }); }); } catch (erro) { listaProdutos.innerHTML = '

Não foi possível carregar o catálogo.

'; mostrarMensagem(erro.message, false); } } form.addEventListener('submit', async (event) => { event.preventDefault(); const dados = { id: document.getElementById('idProduto').value || null, nome: document.getElementById('nome').value, descricao: document.getElementById('descricao').value, preco: document.getElementById('preco').value, estoque: document.getElementById('estoque').value, marca: document.getElementById('marca').value, categoria: document.getElementById('categoria').value, imagem: document.getElementById('imagem').value }; try { const metodo = editandoId ? 'PUT' : 'POST'; const url = editandoId ? `/api/produtos/${editandoId}` : '/api/produtos'; const resposta = await fetch(url, { method: metodo, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(dados) }); const texto = await resposta.text(); if (!resposta.ok) { throw new Error(texto || 'Erro ao salvar produto.'); } mostrarMensagem(editandoId ? 'Produto atualizado com sucesso!' : 'Produto cadastrado com sucesso!'); limparFormulario(); carregarProdutos(); } catch (erro) { mostrarMensagem(erro.message, false); } }); btnCancelar.addEventListener('click', limparFormulario); carregarProdutos();
